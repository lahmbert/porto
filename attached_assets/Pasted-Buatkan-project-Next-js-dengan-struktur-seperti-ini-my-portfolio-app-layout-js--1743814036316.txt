Buatkan project Next.js dengan struktur seperti ini:

/my-portfolio/
├── app/
│   ├── layout.js
│   ├── page.js
│   ├── projects/
│   │   └── page.js
│   ├── about/
│   │   └── page.js
├── components/
│   ├── Navbar.js
│   ├── Footer.js
│   ├── ProjectCard.js
├── public/
│   └── profile.jpg (dummy file)
├── styles/
│   └── globals.css
├── tailwind.config.js
├── next.config.mjs
├── package.json
└── README.md

Instal Tailwind CSS dan setup sesuai dokumentasi resminya.

Lalu isi beberapa file berikut dengan kode ini:

--- app/layout.js ---
export const metadata = {
  title: 'Muhammad Sabran | Portfolio',
  description: 'Fullstack Developer Portfolio',
};
export default function RootLayout({ children }) {
  return (
    <html lang="en">
      <body className="bg-gray-100 text-gray-900">
        <main className="max-w-5xl mx-auto px-4">
          {children}
        </main>
      </body>
    </html>
  );
}

import Link from 'next/link';

--- components/ProjectCard.js ---
export default function ProjectCard({ title, github, live, description }) {
  return (
    <div className="p-4 border rounded-xl shadow-sm bg-white hover:shadow-md transition">
      <h3 className="text-xl font-semibold">{title}</h3>
      <p className="text-sm my-2">{description}</p>
      <div className="flex gap-4 text-sm">
        <a href={github} target="_blank" className="text-blue-600 underline">GitHub</a>
        {live && <a href={live} target="_blank" className="text-green-600 underline">Live Demo</a>}
      </div>
    </div>
  );
}

Instal Tailwind CSS dan isi `styles/globals.css` dengan:
@tailwind base;
@tailwind components;
@tailwind utilities;

Edit tailwind.config.js agar `content` memuat './app/**/*.{js,ts,jsx,tsx}' dan './components/**/*.{js,ts,jsx,tsx}'
